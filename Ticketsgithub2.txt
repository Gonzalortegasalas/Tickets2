function prepareImageForVision(dataUrl, callback) {
  const img = new Image();
  img.onload = () => {
    try {
      // A 2200 px long edge retains receipt text while avoiding oversized API/KV payloads.
      const maxEdge = 2200;
      const scale = Math.min(1, maxEdge / Math.max(img.width, img.height));
      const width = Math.max(1, Math.round(img.width * scale));
      const height = Math.max(1, Math.round(img.height * scale));
      const canvas = document.createElement('canvas');
      canvas.width = width; canvas.height = height;
      const ctx = canvas.getContext('2d');
      ctx.drawImage(img, 0, 0, width, height);
      // Scan pass: improve text/background separation without trimming the ticket.
      enhanceContrast(ctx, width, height);
      callback(canvas.toDataURL('image/jpeg', 0.9));
    } catch(err) {
      console.error('Image preparation failed:', err);
      callback(dataUrl);
    }
  };
  img.onerror = () => callback(dataUrl);
  img.src = dataUrl;
}

function enhanceContrast(ctx, w, h) {
  const imgData = ctx.getImageData(0, 0, w, h);
  const d = imgData.data;

  const histogram = new Array(256).fill(0);
  for(let i = 0; i < d.length; i += 4) {
    const v = Math.round((d[i] + d[i+1] + d[i+2]) / 3);
    histogram[v]++;
  }
  const totalPixels = w * h;
  let cumulative = 0;
  let lowPercentile = 0, highPercentile = 255;
  for(let v = 0; v < 256; v++) {
    cumulative += histogram[v];
    if(cumulative / totalPixels >= 0.02 && lowPercentile === 0) lowPercentile = v;
    if(cumulative / totalPixels >= 0.98) { highPercentile = v; break; }
  }
  const range = Math.max(1, highPercentile - lowPercentile);

  const brightnessBoost = 15;
  for(let i = 0; i < d.length; i += 4) {
    for(let c = 0; c < 3; c++) {
      let val = ((d[i+c] - lowPercentile) / range) * 255 + brightnessBoost;
      d[i+c] = Math.min(255, Math.max(0, val));
    }
  }
  ctx.putImageData(imgData, 0, 0);
}

// ── EXIF ORIENTATION FIX ───────────────────────────────────────────────
function fixImageOrientation(dataUrl, callback) {
  const img = new Image();
  img.onload = () => {
    try {
      const orientation = getExifOrientation(dataUrl);
      if(!orientation || orientation === 1) {
        const canvas = document.createElement('canvas');
        canvas.width = img.width; canvas.height = img.height;
        canvas.getContext('2d').drawImage(img, 0, 0);
        callback(canvas.toDataURL('image/jpeg', 0.92));
        return;
      }

      const canvas = document.createElement('canvas');
      const ctx = canvas.getContext('2d');
      let w = img.width, h = img.height;

      if(orientation >= 5 && orientation <= 8) { canvas.width = h; canvas.height = w; }
      else { canvas.width = w; canvas.height = h; }

      switch(orientation) {
        case 2: ctx.transform(-1, 0, 0, 1, w, 0); break;
        case 3: ctx.transform(-1, 0, 0, -1, w, h); break;
        case 4: ctx.transform(1, 0, 0, -1, 0, h); break;
        case 5: ctx.transform(0, 1, 1, 0, 0, 0); break;
        case 6: ctx.transform(0, 1, -1, 0, h, 0); break;
        case 7: ctx.transform(0, -1, -1, 0, h, w); break;
        case 8: ctx.transform(0, -1, 1, 0, 0, w); break;
        default: break;
      }
      ctx.drawImage(img, 0, 0);
      callback(canvas.toDataURL('image/jpeg', 0.92));
    } catch(err) {
      console.error('Orientation fix failed:', err);
      callback(dataUrl);
    }
  };
  img.onerror = () => callback(dataUrl);
  img.src = dataUrl;
}

function getExifOrientation(dataUrl) {
  try {
    if(!dataUrl || typeof dataUrl !== 'string' || !dataUrl.includes(',')) return 1;
    const base64 = dataUrl.split(',')[1];
    if(!base64 || base64.length < 100) return 1;
    const cleanBase64 = base64.replace(/\\s/g, '');
    const binary = atob(cleanBase64);
    const len = binary.length;
    const buffer = new ArrayBuffer(len);
    const view = new DataView(buffer);
    for(let i = 0; i < len; i++) view.setUint8(i, binary.charCodeAt(i));

    if(view.getUint16(0) !== 0xFFD8) return 1;

    let offset = 2;
    const length = view.byteLength;

    while(offset < length - 1) {
      if(view.getUint8(offset) !== 0xFF) break;
      const marker = view.getUint8(offset + 1);

      if(marker === 0xE1) {
        const exifStart = offset + 4;
        if(view.getUint32(exifStart) !== 0x45786966) return 1;

        const tiffStart = exifStart + 6;
        const little = view.getUint16(tiffStart) === 0x4949;

        const firstIFDOffset = view.getUint32(tiffStart + 4, little);
        const ifdStart = tiffStart + firstIFDOffset;

        const numEntries = view.getUint16(ifdStart, little);

        for(let i = 0; i < numEntries; i++) {
          const entryOffset = ifdStart + 2 + (i * 12);
          const tag = view.getUint16(entryOffset, little);
          if(tag === 0x0112) {
            return view.getUint16(entryOffset + 8, little);
          }
        }
        return 1;
      } else if(marker === 0xD8 || marker === 0xD9 || (marker >= 0xD0 && marker <= 0xD7)) {
        offset += 2;
      } else {
        const segLength = view.getUint16(offset + 2);
        offset += 2 + segLength;
      }
    }
    return 1;
  } catch(e) {
    console.error('EXIF parse error:', e);
    return 1;
  }
}

function rotateImage(which) {
  const isTicket = which === 'ticket';
  const b64 = isTicket ? currentB64 : currentCompB64;
  const mime = isTicket ? currentMime : currentCompMime;
  if(!b64) return;

    img.onload = () => {
      try {
        const canvas = document.createElement('canvas');
        canvas.width = img.height;
        canvas.height = img.width;
        const ctx = canvas.getContext('2d');
        ctx.translate(canvas.width / 2, canvas.height / 2);
        ctx.rotate(90 * Math.PI / 180);
        ctx.drawImage(img, -img.width / 2, -img.height / 2);
        const rotatedDataUrl = canvas.toDataURL('image/jpeg', 0.92);
        const rotatedB64 = rotatedDataUrl.split(',')[1];

        if(!rotatedB64) { alert('No se pudo rotar la imagen. Intenta tomar la foto de nuevo.'); return; }

        if(isTicket) {
          currentB64 = rotatedB64;
          currentMime = 'image/jpeg';
          document.getElementById('preview-ticket').src = rotatedDataUrl;
          if(window._pending) window._pending.imgB64 = currentB64;
        } else {
          currentCompB64 = rotatedB64;
          currentCompMime = 'image/jpeg';
          document.getElementById('preview-comp').src = rotatedDataUrl;
          if(window._pending) window._pending.compB64 = currentCompB64;
        }
      } catch(innerErr) {
        console.error('Error rotando imagen (canvas):', innerErr);
        alert('No se pudo rotar la imagen. Intenta de nuevo.');
      }
    };
    img.onerror = () => {
      alert('No se pudo cargar la imagen para rotarla. Intenta tomar la foto de nuevo.');
    };
    img.src = 'data:' + mime + ';base64,' + b64;
  } catch(err) {
    console.error('Error en rotateImage:', err);
    alert('No se pudo rotar la imagen.');
  }
}

        const diff = Math.abs(totalComp - sumaCheck);
        if(diff > 0.5 && sumaCheck > 0) {
          totalFinal = sumaCheck;
          metodo = 'suma consumo+propina (no coincidía con total leído)';
        } else {
          totalFinal = totalComp;
          metodo = 'total del comprobante';
        }
      } else if(totalComp > 0) {
        totalFinal = totalComp;
        metodo = 'total del comprobante';
      } else if(sumaCheck > 0) {
        totalFinal = sumaCheck;
        metodo = 'suma consumo+propina';
      } else if(totalTk > 0) {
        totalFinal = totalTk;
        metodo = 'total del ticket (comprobante no legible)';
      }

      if(totalFinal > 0) {
        info.total = totalFinal;
        info.notas = (info.notas ? info.notas + ' | ' : '') + \`Monto del comprobante (\${metodo}): \${info.moneda||'$'} \${totalFinal.toFixed(2)}\`;
      }

      delete info.total_comprobante;
      delete info.total_ticket;
      delete info.consumo_comprobante;
      delete info.propina_comprobante;
    }

    if(info.moneda && info.moneda !== 'MXN') {
      try {
        document.getElementById('scan-result').innerHTML = '<div class="card" style="text-align:center;padding:24px"><span class="spinner" style="border-color:rgba(0,122,255,0.2);border-top-color:var(--accent)"></span> Obteniendo tipo de cambio...</div>';
        const fxRes = await fetch(\`/fx/\${info.moneda}\`);
        const fxData = await fxRes.json();
        if(fxData && fxData.rates && fxData.rates.MXN) {
          const rate = fxData.rates.MXN;
          info.moneda_original = info.moneda;
          info.total_original = info.total;
          info.total = Math.round(info.total * rate * 100) / 100;
          info.subtotal = Math.round((info.subtotal || 0) * rate * 100) / 100;
          info.impuestos = Math.round((info.impuestos || 0) * rate * 100) / 100;
          if(info.items) info.items = info.items.map(it => ({...it, precio: Math.round((+it.precio||0) * rate * 100)/100}));
          info.tipo_cambio = Math.round(rate * 10000) / 10000;
          info.moneda = 'MXN';
          info.notas = (info.notas ? info.notas + ' | ' : '') + \`Tipo de cambio \${info.moneda_original}/MXN: \${info.tipo_cambio} (\${new Date().toLocaleDateString('es-MX')})\`;
        }
      } catch(fxErr) {
        info.notas = (info.notas ? info.notas + ' | ' : '') + \`No se pudo obtener tipo de cambio para \${info.moneda}\`;
      }
    }
    if(callId !== _analyzeCallId) { return; }
    showResult(info, null);

  } catch(e) {
    if(callId !== _analyzeCallId) { return; }
    let msg = e.message || 'Error desconocido';
    msg += ' | Detalle técnico: ' + String(e);
    showResult(null, msg);
  } finally {
    if(callId === _analyzeCallId && analyzeBtn) {
      analyzeBtn.disabled = false;
      analyzeBtn.innerHTML = '✦ Analizar con IA';
    }
  }

}

// ── SAVE ───────────────────────────────────────────────────────────────
function saveTicket() {
  if(!window._pending) return;
  const p = window._pending;
  if(tickets.find(t => t.id === p.id)) return;
  const now = new Date().toISOString();
  tickets.unshift({ id: p.id, info: p.info, imgB64: p.imgB64, mime: p.mime, compB64: p.compB64, compMime: p.compMime, savedAt: now, updatedAt: now });
  persistTickets({ upserts: [tickets[0]] });
}

function updateTotal(tid, newValue) {
  const newTotal = parseFloat(newValue);
  if(isNaN(newTotal) || newTotal < 0) {
    alert('Ingresa un monto válido.');
    return;
  }

  const t = tickets.find(x => x.id === tid);
  if(t) {
    t.info.total = newTotal;
    t.updatedAt = new Date().toISOString();
    t.info.notas = (t.info.notas ? t.info.notas + ' | ' : '') + \`Monto editado manualmente a \${t.info.moneda||'$'} \${newTotal.toFixed(2)}\`;
    persistTickets({ upserts: [t] });
  }

  if(window._pending && window._pending.id === tid) {
    window._pending.info.total = newTotal;
  }

  const statusEl = document.getElementById(\`save-status-\${tid}\`);
  if(statusEl) {
    statusEl.textContent = '✓ Monto actualizado';
    statusEl.style.color = 'var(--accent)';
    setTimeout(() => {
      statusEl.textContent = '✓ Guardado automáticamente';
      statusEl.style.color = 'var(--success)';
    }, 1800);
  }

  if(document.getElementById('sec-tickets').classList.contains('active')) renderTickets();
}

// ── DATE EDITING ────────────────────────────────────────────────────────
function fechaToISO(fecha) {
  if(!fecha) return '';
  if(/^\\d{4}-\\d{2}-\\d{2}$/.test(fecha)) return fecha;
  const parts = fecha.split('/');
  if(parts.length !== 3) return '';
  let [d, m, y] = parts;
  if(y.length === 4) return \`\${y}-\${m.padStart(2,'0')}-\${d.padStart(2,'0')}\`;
  return '';
}

function isoToFecha(iso) {
  if(!iso) return '';
  const [y, m, d] = iso.split('-');
  return \`\${d}/\${m}/\${y}\`;
}

function updateFecha(tid, isoValue) {
  if(!isoValue) return;
  const newFecha = isoToFecha(isoValue);

  const t = tickets.find(x => x.id === tid);
  if(t) {
    t.info.fecha = newFecha;
    t.updatedAt = new Date().toISOString();
    t.info.notas = (t.info.notas ? t.info.notas + ' | ' : '') + \`Fecha editada manualmente a \${newFecha}\`;
    persistTickets({ upserts: [t] });
  }

  if(window._pending && window._pending.id === tid) {
    window._pending.info.fecha = newFecha;
  }

  const statusEl = document.getElementById(\`save-status-\${tid}\`);
  if(statusEl) {
    statusEl.textContent = '✓ Fecha actualizada';
    statusEl.style.color = 'var(--accent)';
    setTimeout(() => {
      statusEl.textContent = '✓ Guardado automáticamente';
      statusEl.style.color = 'var(--success)';
    }, 1800);
  }

  if(document.getElementById('sec-tickets').classList.contains('active')) renderTickets();
}

// ── ORIGINAL CURRENCY EDITING ──────────────────────────────────────────
function updateOriginal(tid, newValue) {
  const newOriginal = parseFloat(newValue);
  if(isNaN(newOriginal) || newOriginal < 0) {
    alert('Ingresa un monto válido.');
    return;
  }

  const applyUpdate = (info) => {
    const rate = +info.tipo_cambio || 1;
    info.total_original = newOriginal;
    info.total = Math.round(newOriginal * rate * 100) / 100;
    info.notas = (info.notas ? info.notas + ' | ' : '') + \`Monto original editado a \${info.moneda_original} \${newOriginal.toFixed(2)} -> MXN \${info.total.toFixed(2)} (tipo de cambio \${rate})\`;
  };

  const t = tickets.find(x => x.id === tid);
  if(t) {
    applyUpdate(t.info);
    t.updatedAt = new Date().toISOString();
    persistTickets({ upserts: [t] });
  }

  if(window._pending && window._pending.id === tid) {
    applyUpdate(window._pending.info);
    const totalInput = document.getElementById(\`total-input-\${tid}\`);
    if(totalInput) totalInput.value = window._pending.info.total.toFixed(2);
  } else if(t) {
    const totalInput = document.getElementById(\`total-input-\${tid}\`);
    if(totalInput) totalInput.value = t.info.total.toFixed(2);
  }

  const statusEl = document.getElementById(\`save-status-\${tid}\`);
  if(statusEl) {
    statusEl.textContent = '✓ Monto recalculado';
    statusEl.style.color = 'var(--accent)';
    setTimeout(() => {
      statusEl.textContent = '✓ Guardado automáticamente';
      statusEl.style.color = 'var(--success)';
    }, 1800);
  }

  if(document.getElementById('sec-tickets').classList.contains('active')) renderTickets();
}

// ── PERSISTENCE ─────────────────────────────────────────────────────────
// CRITICAL FIX: previous versions sent the entire in-memory \`tickets\` array on every save
// and the server replaced its whole index with that array. If the in-memory list was ever
// out of sync (a slow network response, two saves racing, a fresh tab that hadn't finished
// loading yet) the next save would permanently delete every ticket missing from that snapshot.
// Now each save explicitly lists which tickets to upsert; nothing is ever deleted unless the
// user explicitly removes that ticket (deleteTicket / clearAll), so a stale in-memory list can
// never wipe out tickets that already exist on the server.
let serverSaveQueue = Promise.resolve();

function setSyncStatus(message, isError = false) {
  const el = document.getElementById('sync-status');
  if(!el) return;
  el.textContent = message;
  el.style.color = isError ? 'var(--danger)' : 'var(--sub)';
}

function localTicketCache() {
  return tickets.map(({imgB64, compB64, ...ticket}) => ({...ticket, imgB64: null, compB64: null}));
}

function persistTickets(options = {}) {
  try {
    localStorage.setItem('tickets_v2', JSON.stringify(localTicketCache()));
  } catch(e) {
    console.error('Error guardando el índice local:', e);
  }

  const upserts = options.upserts ? options.upserts.map(t => ({...t, info: {...t.info}})) : tickets.map(t => ({...t, info: {...t.info}}));
  const deletes = options.deletes || [];

  setSyncStatus('Guardando en Cloudflare…');
  serverSaveQueue = serverSaveQueue
    .then(() => saveToServer(upserts, deletes))
    .then(() => setSyncStatus('✓ Guardado en Cloudflare'))
    .catch(e => {
      console.error('Error guardando en servidor:', e);
      setSyncStatus('No se pudo guardar en Cloudflare. Revisa tu conexión.', true);
    });
  return serverSaveQueue;
}

async function saveToServer(upserts, deletes) {
  const res = await fetch('/kv/save', {
    method: 'POST',
    headers: {'Content-Type':'application/json'},
    body: JSON.stringify({ upserts, deletes })
  });
  const result = await res.json().catch(() => ({}));
  if(!res.ok || !result.ok) throw new Error(result.error || 'Cloudflare rechazó el guardado');
}

async function loadFromServer() {
  try {
    const res = await fetch('/kv/load');
    const data = await res.json();
    if(!res.ok) throw new Error(data.error || 'No se pudieron cargar los tickets');
    if(data && Array.isArray(data.tickets)) {
      const serverById = new Map(data.tickets.map(t => [t.id, t]));
      tickets.forEach(localTicket => {
        const serverTicket = serverById.get(localTicket.id);
        if(!serverTicket || new Date(localTicket.updatedAt || localTicket.savedAt || 0) > new Date(serverTicket.updatedAt || serverTicket.savedAt || 0)) {
          serverById.set(localTicket.id, localTicket);
        }
      });
      tickets = [...serverById.values()].sort((a,b) => new Date(b.savedAt) - new Date(a.savedAt));
      localStorage.setItem('tickets_v2', JSON.stringify(localTicketCache()));
      setSyncStatus(tickets.length ? '✓ Tickets cargados desde Cloudflare' : 'Sin tickets guardados todavía');
      return true;
    }
  } catch(e) {
    console.error('Error cargando desde servidor:', e);
    setSyncStatus('Modo local: Cloudflare no respondió', true);
  }
  return false;
}

// ── PDF ────────────────────────────────────────────────────────────────
function exportPDF(tid) {
  let info, imgB64, mime, compB64, compMime;
  if(!tid && window._pending) {
    info = window._pending.info; imgB64 = window._pending.imgB64; mime = window._pending.mime;
    compB64 = window._pending.compB64; compMime = window._pending.compMime;
  } else {
    const t = tickets.find(x => x.id === tid);
    if(!t) return;
    info = t.info; imgB64 = t.imgB64; mime = t.mime;
    compB64 = t.compB64; compMime = t.compMime;
  }
  loadJsPDF(() => buildPDF(info, imgB64, mime, compB64, compMime));
}

function loadJsPDF(cb) {
  if(window.jspdf) { cb(); return; }
  const s = document.createElement('script');
  s.src = 'https://cdnjs.cloudflare.com/ajax/libs/jspdf/2.5.1/jspdf.umd.min.js';
  s.onload = cb;
  document.head.appendChild(s);
}

function buildPDF(info, imgB64, mime, compB64, compMime) {
  const { jsPDF } = window.jspdf;
  const doc = buildSinglePagePDF(info, imgB64, mime, compB64, compMime);
  const fname = \`ticket_\${(info.tienda||'ticket').replace(/[^a-zA-Z0-9]/g,'_')}_\${(info.fecha||Date.now().toString()).replace(/\\//g,'-')}.pdf\`;
  doc.save(fname);
}

function buildSinglePagePDF(info, imgB64, mime, compB64, compMime) {
  const { jsPDF } = window.jspdf;
  const format = compB64 ? 'a4' : 'a5';
  const doc = new jsPDF({ unit: 'mm', format });
  const W = doc.internal.pageSize.width;
  const H = doc.internal.pageSize.height;
  let y = 18;

  doc.setFontSize(20); doc.setFont('helvetica','bold');
  doc.text(info.tienda || 'Ticket', W/2, y, { align: 'center' }); y += 8;

  doc.setFontSize(10); doc.setFont('helvetica','normal');
  doc.setTextColor(110,110,115);
  doc.text(\`\${info.categoria||''} · \${info.fecha||''}\`, W/2, y, { align: 'center' }); y += 5;
  if(info.direccion) { doc.text(info.direccion, W/2, y, { align: 'center' }); y += 5; }

  doc.setTextColor(0,0,0);
  doc.setDrawColor(210,210,214); doc.line(10, y, W-10, y); y += 8;

  const imagesTop = y;

  if(compB64 && imgB64) {
    const colW = (W - 30) / 2;
    const maxImgH = 95;

    doc.setFontSize(9); doc.setFont('helvetica','bold'); doc.setTextColor(110,110,115);
    doc.text('TICKET', 10 + colW/2, imagesTop, {align:'center'});
    doc.text('COMPROBANTE', 20 + colW + colW/2, imagesTop, {align:'center'});
    doc.setTextColor(0,0,0);
    const imgY = imagesTop + 5;

    try {
      const imgType = (mime||'image/jpeg').split('/')[1].toUpperCase().replace('JPEG','JPG');
      const dataUri = 'data:'+mime+';base64,'+imgB64;
      const props = doc.getImageProperties(dataUri);
      const ratio = Math.min(colW / props.width, maxImgH / props.height);
      const drawW = props.width * ratio, drawH = props.height * ratio;
      const drawX = 10 + (colW - drawW) / 2;
      doc.addImage(dataUri, imgType, drawX, imgY, drawW, drawH, undefined, 'FAST');
    } catch(e) {}

    try {
      const cType = (compMime||'image/jpeg').split('/')[1].toUpperCase().replace('JPEG','JPG');
      const cDataUri = 'data:'+compMime+';base64,'+compB64;
      const cProps = doc.getImageProperties(cDataUri);
      const cRatio = Math.min(colW / cProps.width, maxImgH / cProps.height);
      const drawCW = cProps.width * cRatio, drawCH = cProps.height * cRatio;
      const drawCX = 20 + colW + (colW - drawCW) / 2;
      doc.addImage(cDataUri, cType, drawCX, imgY, drawCW, drawCH, undefined, 'FAST');
    } catch(e) {}

    y = imgY + maxImgH + 10;

  } else if(imgB64) {
    try {
      const imgType = (mime||'image/jpeg').split('/')[1].toUpperCase().replace('JPEG','JPG');
      const dataUri = 'data:'+mime+';base64,'+imgB64;
      const props = doc.getImageProperties(dataUri);
      const maxW = W - 20, maxH = 70;
      const ratio = Math.min(maxW / props.width, maxH / props.height);
      const drawW = props.width * ratio, drawH = props.height * ratio;
      const drawX = 10 + (maxW - drawW) / 2;
      doc.addImage(dataUri, imgType, drawX, y, drawW, drawH, undefined, 'FAST');
      y += drawH + 8;
    } catch(e) {}
  }

  doc.setFontSize(10);
  if(info.items && info.items.length) {
    doc.setFont('helvetica','bold'); doc.text('Productos', 10, y); y += 5;
    doc.setFont('helvetica','normal');
    info.items.forEach(it => {
      if(y > H - 20) { doc.addPage(); y = 15; }
      doc.text(String(it.nombre||''), 10, y);
      doc.text(\`\${info.moneda||'$'} \${(+it.precio||0).toFixed(2)}\`, W-10, y, { align: 'right' });
      y += 5;
    });
    doc.setDrawColor(210,210,214); doc.line(10, y, W-10, y); y += 6;
  }

  doc.setFont('helvetica','bold'); doc.setFontSize(13);
  doc.text('TOTAL', 10, y);
  doc.text(\`\${info.moneda||'$'} \${(+info.total||0).toFixed(2)}\`, W-10, y, { align: 'right' });

  return doc;
}

function restoreTickets(e) {
  const f = e.target.files[0]; if(!f) return;
  const r = new FileReader();
  r.onload = () => {
    try {
      const restored = JSON.parse(r.result);
      if(!Array.isArray(restored)) throw new Error('Formato inválido');
      const existingIds = new Set(tickets.map(t => t.id));
      const newOnes = restored.filter(t => !existingIds.has(t.id));
      tickets = [...newOnes, ...tickets];
      persistTickets({ upserts: newOnes });
      alert(\`\${newOnes.length} ticket(s) restaurado(s).\`);
      renderTickets();
    } catch(err) {
      alert('Error al restaurar: archivo inválido.');
    }
  };
  r.readAsText(f);
}

function deleteTicket(id) {
  const t = tickets.find(x => x.id === id);
  const nombre = t ? t.info.tienda : (window._pending && window._pending.id === id ? window._pending.info.tienda : null);
  if(confirm(\`¿Eliminar el ticket de "\${nombre || 'este ticket'}"? No se puede deshacer.\`)) {
    tickets = tickets.filter(x => x.id !== id);
    persistTickets({ upserts: [], deletes: [id] });
    renderTickets();
    const card = document.getElementById(id);
    if(card) card.closest('.card')?.remove();
    if(window._pending && window._pending.id === id) {
      document.getElementById('scan-result').innerHTML = '';
      clearTicket();
    }
  }
}

function rotateManualFoto() {
  if(!_manualFotoB64) return;
  try {
    const img = new Image();
    img.onload = () => {
      try {
        const canvas = document.createElement('canvas');
        canvas.width = img.height;
        canvas.height = img.width;
        const ctx = canvas.getContext('2d');
        ctx.translate(canvas.width / 2, canvas.height / 2);
        ctx.rotate(90 * Math.PI / 180);
        ctx.drawImage(img, -img.width / 2, -img.height / 2);
        const rotatedDataUrl = canvas.toDataURL('image/jpeg', 0.92);
        const rotatedB64 = rotatedDataUrl.split(',')[1];
        if(rotatedB64) {
          _manualFotoB64 = rotatedB64;
          _manualFotoMime = 'image/jpeg';
          document.getElementById('manual-foto-preview').src = rotatedDataUrl;
        }
      } catch(e) { console.error(e); }
    };
    img.src = 'data:' + _manualFotoMime + ';base64,' + _manualFotoB64;
  } catch(e) { console.error(e); }
}

function parseFecha(fecha) {
  if(!fecha) return {anio:1900, mes:1, dia:1};
  let parts;
  if(fecha.includes('/')) parts = fecha.split('/');
  else if(fecha.includes('-')) parts = fecha.split('-').reverse();
  else return {anio:1900, mes:1, dia:1};
  if(parts[0].length === 4) parts = parts.reverse();
  return { dia: parseInt(parts[0])||1, mes: parseInt(parts[1])||1, anio: parseInt(parts[2])||1900 };
}

function parseHora(hora) {
  if(!hora) return {horas:0, mins:0};
  const parts = hora.split(':');
  return { horas: parseInt(parts[0])||0, mins: parseInt(parts[1])||0 };
}

function nombreCorto(tienda) {
  if(!tienda) return '';
  let n = tienda.split(',')[0].split('.')[0].trim();
  return n.substring(0, 12).trim();
}

function getCuenta(info) {
  if(info.cuenta_override) return info.cuenta_override;

  const tarjeta = (info.tarjeta || '').toString().trim();
  if(/^\\d{4}$/.test(tarjeta)) return tarjeta;

  const allText = JSON.stringify(info);
  const cardMatch = allText.match(/\\*{2,}(\\d{4})/) || allText.match(/\\b3139\\b/);
  if(cardMatch) return cardMatch[1] || '3139';

  return 'EFVO';
}

function doExport() {
  const wb = XLSX.utils.book_new();

  const headers = [
    'Colaborador', 'Fecha', 'Hora',
    'Establecimiento y descripcion corta', 'Monto (MXN)', 'Cuenta***',
    'Año', 'Mes', 'Dia', 'Hora_', 'Min', 'Nombre corto', 'Monto_aux', 'Cuenta_aux',
    'Codificacion'
  ];

  const rows = [headers];

  tickets.forEach(t => {
    const info = t.info;
    const f = parseFecha(info.fecha);
    const h = parseHora(info.hora || '');
    const cuenta = getCuenta(info);
    const monto = +info.total || 0;
    const nombre = nombreCorto(info.tienda);
    const montoStr = monto % 1 === 0 ? monto.toFixed(0) : monto.toFixed(2);
    const horaStr = info.hora ? \`\${h.horas}.\${String(h.mins).padStart(2,'0')}\` : '0.00';
    const codificacion = \`GOS [\${cuenta}] \${f.anio}.\${String(f.mes).padStart(2,'0')}.\${String(f.dia).padStart(2,'0')} - \${horaStr}HRS - \${nombre} - MXN$ \${montoStr}\`;

    rows.push([
      'GOS',
      info.fecha || '',
      info.hora || '',
      \`\${info.tienda||''} (\${info.categoria||'Otro'})\`,
      monto,
      cuenta,
      f.anio, f.mes, f.dia, h.horas, h.mins, nombre, monto, cuenta,
      codificacion
    ]);
  });

  const ws = XLSX.utils.aoa_to_sheet(rows);
  ws['!cols'] = [
    {wch:12},{wch:12},{wch:8},{wch:35},{wch:12},{wch:10},
    {wch:6},{wch:5},{wch:5},{wch:6},{wch:5},{wch:14},{wch:10},{wch:10},
    {wch:60}
  ];
  XLSX.utils.book_append_sheet(wb, ws, 'Gastos');

  const bycat = {};
  const bycatN = {};
  tickets.forEach(t => {
    const c = t.info.categoria || 'Otro';
    bycat[c] = (bycat[c]||0) + (+t.info.total||0);
    bycatN[c] = (bycatN[c]||0) + 1;
  });
  const sumRows = [['Categoría','Total MXN','# Tickets']];
  Object.entries(bycat).sort((a,b)=>b[1]-a[1]).forEach(([c,v]) => {
    sumRows.push([c, v, bycatN[c]]);
  });
  sumRows.push(['','','']);
  sumRows.push(['TOTAL', tickets.reduce((a,t)=>a+(+t.info.total||0),0), tickets.length]);
  const ws2 = XLSX.utils.aoa_to_sheet(sumRows);
  ws2['!cols'] = [{wch:20},{wch:12},{wch:10}];
  XLSX.utils.book_append_sheet(wb, ws2, 'Resumen');

  const fecha = new Date().toISOString().split('T')[0];
  XLSX.writeFile(wb, \`gastos_GOS_\${fecha}.xlsx\`);
}

function getMonday(d) {
  const date = new Date(d);
  const day = date.getDay();
  const diff = date.getDate() - day + (day === 0 ? -6 : 1);
  return new Date(date.setDate(diff));
}

function parseFechaToDate(fecha) {
  const f = parseFecha(fecha);
  return new Date(f.anio, f.mes - 1, f.dia);
}

function fmtDate(d) {
  return d.toISOString().split('T')[0];
}

function sanitizeFolderName(name) {
  return name.replace(/[<>:"/\\\\|?*]/g, '-').substring(0, 150);
}

async function exportZIP() {
  if(!tickets.length) { alert('No hay tickets guardados para exportar.'); return; }
  const btn = document.querySelector('button[onclick="exportZIP()"]');
  if(btn) { btn.disabled = true; btn.textContent = 'Cargando librerías...'; }

  try {
    if(!window.JSZip) await loadScript('https://cdnjs.cloudflare.com/ajax/libs/jszip/3.10.1/jszip.min.js');

    if(btn) btn.textContent = 'Generando carpetas...';

    const zip = new JSZip();

    const weeks = {};
    tickets.forEach(t => {
      const d = parseFechaToDate(t.info.fecha);
      const monday = getMonday(new Date(d));
      const sunday = new Date(monday); sunday.setDate(monday.getDate() + 6);
      const weekKey = \`Tickets_\${fmtDate(monday)}_al_\${fmtDate(sunday)}\`;
      if(!weeks[weekKey]) weeks[weekKey] = [];
      weeks[weekKey].push(t);
    });

    for(const weekFolder in weeks) {
      const weekZip = zip.folder(weekFolder);
      const weekTickets = weeks[weekFolder];

      for(const t of weekTickets) {
        const info = t.info;
        const f = parseFecha(info.fecha);
        const h = parseHora(info.hora || '');
        const cuenta = getCuenta(info);
        const monto = +info.total || 0;
        const nombre = nombreCorto(info.tienda);

function loadScript(src) {
  return new Promise((resolve, reject) => {
    const s = document.createElement('script');
    s.src = src;
    s.onload = resolve;
    s.onerror = () => reject(new Error('No se pudo cargar: ' + src));
    document.head.appendChild(s);
  });
}

function buildPDFBlob(info, imgB64, mime, compB64, compMime) {
  return new Promise((resolve) => {
    const doc = buildSinglePagePDF(info, imgB64, mime, compB64, compMime);
    resolve(doc.output('blob'));
  });

      }
      try {
        const body = await request.json();
        const apiKey = request.headers.get('x-api-key');
        const res = await fetch('https://api.openai.com/v1/responses', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + apiKey },
          body: JSON.stringify(body)

      const currency = url.pathname.replace('/fx/', '').toUpperCase();

        const fxData = await fxRes.json();

      }
    }

    // ── KV SAVE ───────────────────────────────────────────────────────────
    // CRITICAL FIX: this endpoint used to receive the client's FULL in-memory ticket array
    // and treat it as the complete source of truth, deleting anything from KV not present in
    // that snapshot. Any race (slow request, page that hadn't finished loading all tickets yet,
    // two saves overlapping) could wipe out tickets that were never meant to be deleted.
    //
    // Now the client sends two explicit lists:
    //   - upserts: tickets to create or update (always safe, never deletes anything else)
    //   - deletes: ticket ids the user explicitly removed
    // Every other ticket already in KV is left completely untouched.
    if (url.pathname === '/kv/save' && request.method === 'POST') {
      try {
        const body = await request.json();
        const upserts = Array.isArray(body.upserts) ? body.upserts : [];
        const deletes = Array.isArray(body.deletes) ? body.deletes : [];

        for (const ticket of upserts) {
          if (!ticket || !ticket.id) continue;
          await env.TICKETS_KV.put(`ticket:${ticket.id}`, JSON.stringify(ticket));
        }
        for (const id of deletes) {
          if (!id) continue;
          await env.TICKETS_KV.delete(`ticket:${id}`);
        }

        // Maintain an index of known ticket ids so /kv/load can list them without a KV "list" call per request.
        const indexRaw = await env.TICKETS_KV.get('tickets_gos_index');
        let index = indexRaw ? JSON.parse(indexRaw) : [];
        const indexSet = new Set(index);
        upserts.forEach(t => t && t.id && indexSet.add(t.id));
        deletes.forEach(id => id && indexSet.delete(id));
        await env.TICKETS_KV.put('tickets_gos_index', JSON.stringify([...indexSet]));

        return new Response(JSON.stringify({ok: true}), { headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' } });
      } catch(e) {
        return new Response(JSON.stringify({error: e.message}), { status: 500, headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' } });
      }
    }

    // ── KV LOAD ───────────────────────────────────────────────────────────
    if (url.pathname === '/kv/load' && request.method === 'GET') {
      try {
        const indexRaw = await env.TICKETS_KV.get('tickets_gos_index');
        let tickets = [];

        if (indexRaw) {
          const ids = JSON.parse(indexRaw);
          const records = await Promise.all(ids.map(id => env.TICKETS_KV.get(`ticket:${id}`)));
          tickets = records.filter(Boolean).map(r => JSON.parse(r));
        } else {
          // One-time migration path for installs that still have the old single-blob format.
          const legacy = await env.TICKETS_KV.get('tickets_gos');
          if (legacy) {
            tickets = JSON.parse(legacy);
            // Migrate forward immediately so future saves use the safe per-ticket format.
            await Promise.all(tickets.map(t => t && t.id ? env.TICKETS_KV.put(`ticket:${t.id}`, JSON.stringify(t)) : null));
            await env.TICKETS_KV.put('tickets_gos_index', JSON.stringify(tickets.filter(t => t && t.id).map(t => t.id)));
          }
        }

        return new Response(JSON.stringify({tickets}), { headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' } });
      } catch(e) {
        return new Response(JSON.stringify({tickets: [], error: e.message}), { headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' } });
      }
    }

    return new Response('Not found', { status: 404 });
  }
